🏓 Construindo o Pong, passo a passo

O mesmo Pong da Aula 2, desmontado em pedaços. Cada passo mostra o código completo daquele estágio do jogo — cole ele inteiro na aba JAVASCRIPT do seu compilador e clique em Executar pra ver o resultado antes de ir pro próximo.

💡 No seu compilador, o HTML já vem pronto com <canvas id="jogo" width="800" height="500"> — não precisa mudar a aba HTML BASE em nenhum passo, só a aba JAVASCRIPT. Os códigos abaixo já usam esse mesmo id="jogo". As prévias nesta página usam canvases menores só pra caber várias na tela, mas a lógica é idêntica ao código que você vai copiar.

As raquetes (e uma função desenhar())

Daqui pra frente, tudo entra dentro de uma função desenhar() — porque no próximo passo vamos chamar ela várias vezes por segundo pra animar.

JavaScript
const canvas = document.getElementById('jogo');
const ctx = canvas.getContext('2d');

const RAQUETE_W = 12, RAQUETE_H = 90, MARGEM = 20;
const r1 = { x: MARGEM, y: canvas.height / 2 - RAQUETE_H / 2, w: RAQUETE_W, h: RAQUETE_H };
const r2 = { x: canvas.width - MARGEM - RAQUETE_W, y: canvas.height / 2 - RAQUETE_H / 2, w: RAQUETE_W, h: RAQUETE_H };

function desenhar() {
  ctx.fillStyle = '#050510';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  ctx.setLineDash([10, 10]);
  ctx.strokeStyle = '#1e1e3a';
  ctx.lineWidth = 2;
  ctx.beginPath();
  ctx.moveTo(canvas.width / 2, 0);
  ctx.lineTo(canvas.width / 2, canvas.height);
  ctx.stroke();
  ctx.setLineDash([]);

  ctx.fillStyle = '#60a5fa';
  ctx.fillRect(r1.x, r1.y, r1.w, r1.h);

  ctx.fillStyle = '#f59e0b';
  ctx.fillRect(r2.x, r2.y, r2.w, r2.h);
}

desenhar();

Movimento — o game loop

Com requestAnimationFrame, a bola anda sozinha e quica nas quatro paredes. Repare que ela ainda atravessa as raquetes — isso só é corrigido no passo 7.

JavaScript
const canvas = document.getElementById('jogo');
const ctx = canvas.getContext('2d');

const RAQUETE_W = 12, RAQUETE_H = 90, MARGEM = 20;
const r1 = { x: MARGEM, y: canvas.height / 2 - RAQUETE_H / 2, w: RAQUETE_W, h: RAQUETE_H };
const r2 = { x: canvas.width - MARGEM - RAQUETE_W, y: canvas.height / 2 - RAQUETE_H / 2, w: RAQUETE_W, h: RAQUETE_H };
const bola = { x: canvas.width / 2, y: canvas.height / 2, r: 8, velX: 5, velY: 3 };

function desenhar() {
  ctx.fillStyle = '#050510';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  ctx.setLineDash([10, 10]);
  ctx.strokeStyle = '#1e1e3a';
  ctx.lineWidth = 2;
  ctx.beginPath();
  ctx.moveTo(canvas.width / 2, 0);
  ctx.lineTo(canvas.width / 2, canvas.height);
  ctx.stroke();
  ctx.setLineDash([]);

  ctx.fillStyle = '#60a5fa';
  ctx.fillRect(r1.x, r1.y, r1.w, r1.h);

  ctx.fillStyle = '#f59e0b';
  ctx.fillRect(r2.x, r2.y, r2.w, r2.h);

  ctx.fillStyle = '#ffffff';
  ctx.beginPath();
  ctx.arc(bola.x, bola.y, bola.r, 0, Math.PI * 2);
  ctx.fill();
}

function gameLoop() {
  requestAnimationFrame(gameLoop);

  bola.x += bola.velX;
  bola.y += bola.velY;

  if (bola.y - bola.r <= 0 || bola.y + bola.r >= canvas.height) bola.velY *= -1;
  if (bola.x - bola.r <= 0 || bola.x + bola.r >= canvas.width) bola.velX *= -1;

  desenhar();
}

6

Controlando as raquetes pelo teclado

Um objeto teclas guarda quais botões estão pressionados. A cada frame, se W estiver pressionado, a raquete 1 sobe.

JavaScript
const canvas = document.getElementById('jogo');
const ctx = canvas.getContext('2d');

const RAQUETE_W = 12, RAQUETE_H = 90, MARGEM = 20, VR = 7;
const r1 = { x: MARGEM, y: canvas.height / 2 - RAQUETE_H / 2, w: RAQUETE_W, h: RAQUETE_H };
const r2 = { x: canvas.width - MARGEM - RAQUETE_W, y: canvas.height / 2 - RAQUETE_H / 2, w: RAQUETE_W, h: RAQUETE_H };
const bola = { x: canvas.width / 2, y: canvas.height / 2, r: 8, velX: 5, velY: 3 };

const teclas = {};
window.addEventListener('keydown', e => { teclas[e.key] = true; e.preventDefault(); });
window.addEventListener('keyup',   e => { teclas[e.key] = false; });

function desenhar() {
  ctx.fillStyle = '#050510';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  ctx.setLineDash([10, 10]);
  ctx.strokeStyle = '#1e1e3a';
  ctx.lineWidth = 2;
  ctx.beginPath();
  ctx.moveTo(canvas.width / 2, 0);
  ctx.lineTo(canvas.width / 2, canvas.height);
  ctx.stroke();
  ctx.setLineDash([]);

  ctx.fillStyle = '#60a5fa';
  ctx.fillRect(r1.x, r1.y, r1.w, r1.h);

  ctx.fillStyle = '#f59e0b';
  ctx.fillRect(r2.x, r2.y, r2.w, r2.h);

  ctx.fillStyle = '#ffffff';
  ctx.beginPath();
  ctx.arc(bola.x, bola.y, bola.r, 0, Math.PI * 2);
  ctx.fill();
}

function gameLoop() {
  requestAnimationFrame(gameLoop);

  if ((teclas['w'] || teclas['W']) && r1.y > 0) r1.y -= VR;
  if ((teclas['s'] || teclas['S']) && r1.y + RAQUETE_H < canvas.height) r1.y += VR;
  if (teclas['ArrowUp']   && r2.y > 0) r2.y -= VR;
  if (teclas['ArrowDown'] && r2.y + RAQUETE_H < canvas.height) r2.y += VR;

  bola.x += bola.velX;
  bola.y += bola.velY;

  if (bola.y - bola.r <= 0 || bola.y + bola.r >= canvas.height) bola.velY *= -1;
  if (bola.x - bola.r <= 0 || bola.x + bola.r >= canvas.width) bola.velX *= -1;

  desenhar();
}

gameLoop();
🖱️ Clique no quadro e use W/S (azul) e ↑/↓ (amarela)
7

Colisão bola × raquete

A função colide() checa se a bola tocou o retângulo da raquete. Se sim, inverte a velocidade em X (e acelera um pouco). Quando erra, resetBola() devolve a bola pro centro.

JavaScript
const canvas = document.getElementById('jogo');
const ctx = canvas.getContext('2d');

const RAQUETE_W = 12, RAQUETE_H = 90, MARGEM = 20, VR = 7;
const r1 = { x: MARGEM, y: canvas.height / 2 - RAQUETE_H / 2, w: RAQUETE_W, h: RAQUETE_H };
const r2 = { x: canvas.width - MARGEM - RAQUETE_W, y: canvas.height / 2 - RAQUETE_H / 2, w: RAQUETE_W, h: RAQUETE_H };
const bola = { x: canvas.width / 2, y: canvas.height / 2, r: 8, velX: 5, velY: 3 };

const teclas = {};
window.addEventListener('keydown', e => { teclas[e.key] = true; e.preventDefault(); });
window.addEventListener('keyup',   e => { teclas[e.key] = false; });

function colide(r) {
  return bola.x - bola.r <= r.x + r.w && bola.x + bola.r >= r.x &&
         bola.y + bola.r >= r.y && bola.y - bola.r <= r.y + r.h;
}

function desenhar() {
  ctx.fillStyle = '#050510';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  ctx.setLineDash([10, 10]);
  ctx.strokeStyle = '#1e1e3a';
  ctx.lineWidth = 2;
  ctx.beginPath();
  ctx.moveTo(canvas.width / 2, 0);
  ctx.lineTo(canvas.width / 2, canvas.height);
  ctx.stroke();
  ctx.setLineDash([]);

  ctx.fillStyle = '#60a5fa';
  ctx.fillRect(r1.x, r1.y, r1.w, r1.h);

  ctx.fillStyle = '#f59e0b';
  ctx.fillRect(r2.x, r2.y, r2.w, r2.h);

  ctx.fillStyle = '#ffffff';
  ctx.beginPath();
  ctx.arc(bola.x, bola.y, bola.r, 0, Math.PI * 2);
  ctx.fill();
}

function resetBola() {
  bola.x = canvas.width / 2; bola.y = canvas.height / 2;
  bola.velX = (Math.random() > 0.5 ? 1 : -1) * 5;
  bola.velY = (Math.random() - 0.5) * 6;
}

function gameLoop() {
  requestAnimationFrame(gameLoop);

  if ((teclas['w'] || teclas['W']) && r1.y > 0) r1.y -= VR;
  if ((teclas['s'] || teclas['S']) && r1.y + RAQUETE_H < canvas.height) r1.y += VR;
  if (teclas['ArrowUp']   && r2.y > 0) r2.y -= VR;
  if (teclas['ArrowDown'] && r2.y + RAQUETE_H < canvas.height) r2.y += VR;

  bola.x += bola.velX;
  bola.y += bola.velY;

  if (bola.y - bola.r <= 0 || bola.y + bola.r >= canvas.height) bola.velY *= -1;

  if (colide(r1) && bola.velX < 0) bola.velX = Math.min(Math.abs(bola.velX) * 1.05, 14);
  if (colide(r2) && bola.velX > 0) bola.velX = -Math.min(Math.abs(bola.velX) * 1.05, 14);

  if (bola.x < 0 || bola.x > canvas.width) resetBola();

  desenhar();
}

gameLoop();
🖱️ Clique no quadro e defenda com W/S e ↑/↓
🏆

Placar e vitória — o jogo completo

Duas variáveis pts1/pts2 contam os pontos. Ao chegar em LIMITE, o jogo trava em fimDeJogo e mostra o vencedor. Este é o Pong da Aula 2.

JavaScript
const canvas = document.getElementById('jogo');
const ctx = canvas.getContext('2d');

const RAQUETE_W = 12, RAQUETE_H = 90, MARGEM = 20, VR = 7, LIMITE = 7;
const r1 = { x: MARGEM, y: canvas.height / 2 - RAQUETE_H / 2, w: RAQUETE_W, h: RAQUETE_H };
const r2 = { x: canvas.width - MARGEM - RAQUETE_W, y: canvas.height / 2 - RAQUETE_H / 2, w: RAQUETE_W, h: RAQUETE_H };
const bola = { x: canvas.width / 2, y: canvas.height / 2, r: 8, velX: 5, velY: 3 };
let pts1 = 0, pts2 = 0, fimDeJogo = false;

const teclas = {};
window.addEventListener('keydown', e => { teclas[e.key] = true; e.preventDefault(); });
window.addEventListener('keyup',   e => { teclas[e.key] = false; });

function colide(r) {
  return bola.x - bola.r <= r.x + r.w && bola.x + bola.r >= r.x &&
         bola.y + bola.r >= r.y && bola.y - bola.r <= r.y + r.h;
}

function desenhar() {
  ctx.fillStyle = '#050510';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  ctx.setLineDash([10, 10]);
  ctx.strokeStyle = '#1e1e3a';
  ctx.lineWidth = 2;
  ctx.beginPath();
  ctx.moveTo(canvas.width / 2, 0);
  ctx.lineTo(canvas.width / 2, canvas.height);
  ctx.stroke();
  ctx.setLineDash([]);

  ctx.fillStyle = '#60a5fa';
  ctx.fillRect(r1.x, r1.y, r1.w, r1.h);

  ctx.fillStyle = '#f59e0b';
  ctx.fillRect(r2.x, r2.y, r2.w, r2.h);

  ctx.fillStyle = '#ffffff';
  ctx.beginPath();
  ctx.arc(bola.x, bola.y, bola.r, 0, Math.PI * 2);
  ctx.fill();

  ctx.fillStyle = '#e2e8f0';
  ctx.font = 'bold 32px sans-serif';
  ctx.textAlign = 'center';
  ctx.fillText(pts1 + '   ' + pts2, canvas.width / 2, 50);

  if (fimDeJogo) {
    const vencedor = pts1 >= LIMITE ? 'Jogador 1 🔵' : 'Jogador 2 🟡';
    ctx.fillStyle = 'rgba(5,5,16,0.85)';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 28px sans-serif';
    ctx.fillText('🏆 ' + vencedor + ' venceu!', canvas.width / 2, canvas.height / 2);
  }
}

function resetBola() {
  bola.x = canvas.width / 2; bola.y = canvas.height / 2;
  bola.velX = (Math.random() > 0.5 ? 1 : -1) * 5;
  bola.velY = (Math.random() - 0.5) * 6;
}

function gameLoop() {
  requestAnimationFrame(gameLoop);
  if (fimDeJogo) { desenhar(); return; }

  if ((teclas['w'] || teclas['W']) && r1.y > 0) r1.y -= VR;
  if ((teclas['s'] || teclas['S']) && r1.y + RAQUETE_H < canvas.height) r1.y += VR;
  if (teclas['ArrowUp']   && r2.y > 0) r2.y -= VR;
  if (teclas['ArrowDown'] && r2.y + RAQUETE_H < canvas.height) r2.y += VR;

  bola.x += bola.velX;
  bola.y += bola.velY;

  if (bola.y - bola.r <= 0 || bola.y + bola.r >= canvas.height) bola.velY *= -1;

  if (colide(r1) && bola.velX < 0) bola.velX = Math.min(Math.abs(bola.velX) * 1.05, 14);
  if (colide(r2) && bola.velX > 0) bola.velX = -Math.min(Math.abs(bola.velX) * 1.05, 14);

  if (bola.x < 0) { pts2++; if (pts2 >= LIMITE) fimDeJogo = true; resetBola(); }
  if (bola.x > canvas.width) { pts1++; if (pts1 >= LIMITE) fimDeJogo = true; resetBola(); }

  desenhar();
}

gameLoop();
🖱️ Clique no quadro — primeiro a 5 pontos vence, depois reinicia sozinho
💡 O que fica desse exercício
Um jogo "complexo" quase sempre é uma pilha de pedaços simples: desenhar → mover → ler entrada do jogador → detectar colisão → marcar pontos. Construir na ordem certa, um pedaço testável de cada vez, é o que faz o Pong parecer fácil.
